            has_global_flag = acot_phanon_forbidden

            has_global_flag = voidark_phanon_vault_cracked
        }
        has_technology = tech_voidark_vault_cracker

    possible = {
        inline_script = megastructures/system_ownership_or_waystation_check
        hidden_trigger = {
            is_phanon_rival_path = no
            acot_phanon_crisis_resolved = no
        }
    }

    placement_rules = {

            custom_tooltip = {

                is_acot_pmc_vault = yes
            }
        }
    }

    on_build_start = {
        from = {
            set_country_flag = is_currently_building_vault_cracker

    }
    on_build_cancel = {
        from = {
            remove_country_flag = is_currently_building_vault_cracker
        }

                limit = {
                    is_acot_pmc_vault = yes
                }
                set_carrier_flag = has_vault_cracker_0
                save_event_target_as = voidark_phanon_vault
            }
        }
        from = {
            remove_country_flag = is_currently_building_vault_cracker
            set_country_flag = built_vault_cracker_0
            country_event = {
                id = voidark.1 #notification event
            }
        }
    }
    ai_weight = {
        weight = 0 #no AI for now
    }
