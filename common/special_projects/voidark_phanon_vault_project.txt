special_project = {
    key = "VOIDARK_PHANON_VAULT_INVASION"
    days_to_research = 360 #tweak
    timelimit = 1800
    picture = GFX_evt_laser_grid
    icon = "gfx/interface/icons/situation_log/situation_log_main_quest.dds"
    location = yes
    remove_when_completed = yes

    event_scope = planet_event
    requirements = {
    }


    on_start = {
        country_event = {
            id = voidark_vault_invasion.1
            days = 1
        }
    }

    on_success = {
        random_system_planet = {
            limit = {
                OR = {
                    is_planet_class = pc_acot_pmc_vault_unshielded
                    is_planet_class = pc_acot_pmc_vault
                }
            }
            hidden_effect = {
                set_carrier_flag = ongoing_vault_infiltration
                add_modifier = {
                    modifier = acot_modifier_vault_delay
                    months = 6
                }
                if = {
                    limit = {
                        NOT = {
                            has_global_flag = acot_phanon_rival_plan_initiated
                        }
                    }
                    if = {
                        limit = {
                            owner? = {
                                NOT = {
                                    has_country_flag = acot_phanon_tech_stage_one_enabled
                                }
                            }
                        }
                        carrier_event = {
                            id = acot_pmc.12
                        }
                    }
                    if = {
                        limit = {
                            owner? = {
                                has_country_flag = acot_phanon_tech_stage_one_enabled
                                NOT = {
                                    has_country_flag = acot_phanon_tech_stage_two_enabled
                                }
                            }
                        }
                        carrier_event = {
                            id = acot_pmc.13
                        }
                    }
                    if = {
                        limit = {
                            owner? = {
                                has_country_flag = acot_phanon_tech_stage_two_enabled
                                NOT = {
                                    has_country_flag = acot_phanon_tech_stage_three_enabled
                                }
                            }
                        }
                        carrier_event = {
                            id = acot_pmc.14
                        }
                    }
                    if = {
                        limit = {
                            owner? = {
                                has_country_flag = acot_phanon_tech_stage_three_enabled
                                NOT = {
                                    has_country_flag = acot_phanon_tech_stage_four_enabled
                                }
                            }
                        }
                        carrier_event = {
                            id = acot_pmc.15
                        }
                    }
                }
                else = {
                    carrier_event = {
                        id = acot_pmc.16
                    }
                }
                create_vault_defense_drones = yes
            }
        }
    }

    on_cancel = {
        country_event = {
            id = voidark_vault_invasion.3
        }
    }
    abort_trigger = {
        any_system = {
            any_system_planet = {
                AND = {
                    OR = {
                        is_planet_class = pc_acot_pmc_vault
                        is_planet_class = pc_acot_pmc_vault_unshielded
                    }
                    OR = {
                        has_carrier_flag = acot_vault_floor_four_cleared
                        has_carrier_flag = acot_vault_rival_cleared
                    }
                }
            }
        }
    }
}
