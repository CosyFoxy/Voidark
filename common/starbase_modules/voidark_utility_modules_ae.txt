    icon = "GFX_voidark_beastport_ae"

    category = sm_naval_support_cat

    inline_script = {
        script = voidark/modules/modules_utility
        ACOT_TECH = "tech_voidark_alpha"
        TIER = alpha
        OTHER = "OR = {
				has_technology = tech_alien_cloning
				is_beastmasters_empire = yes
			}"
    }

    possible = {
        inline_script = voidark/modules/count_engineering_modules
    }

    abort_construction_trigger = {
        has_any_engineering_bay = yes
    }
    resources = {
        category = starbase_arkship_upgrades
        cost = {
            inline_script = voidark/modules/module_costs/module_cost_small_alpha
        }
        upkeep = {
            inline_script = voidark/modules/module_upkeep/module_upkeep_util_alpha
        }
    }
    planet_modifier = {
        planet_building_capacity_add = 3
    }

    country_modifier = {
        country_vivarium_max_capacity_add = 60
    }
    on_finished = {
        set_ship_construction_type = starbase_beastport
    }
    on_destroyed = {
        if = {
            limit = { has_any_beastport = no }
            set_ship_construction_type = starbase_shipyard
        }
    }
}
