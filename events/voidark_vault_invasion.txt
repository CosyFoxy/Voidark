namespace = voidark_vault_invasion

carrier_event = {
    id = voidark_vault_invasion.1 #spawn special project and inform - FIRST CALL
    is_triggered_only = yes
    hide_window = yes

    trigger = {
        NOT = { has_global_flag = aot_phanon_content_OFF }
        OR = {
            is_planet_class = pc_acot_pmc_vault_unshielded
            is_planet_class = pc_acot_pmc_vault
        }
        fromfrom = {
            is_nomadic = yes
            is_ai = no
            OR = {
                is_same_value = event_target:acot_phanon_original_aggressor
                is_same_value = event_target:acot_phanon_real_spook_country
            }
        }
        from = {
            has_country_flag = nomad_vassal@fromfrom
            any_owned_planet = {
                is_planet_class = pc_acot_pmc_vault
            }
        }
    }

    immediate = {
        fromfrom = {
            country_event = {
                id = voidark_vault_invasion.11
            }
        }
    }
}

country_event = {
    id = voidark_vault_invasion.11 #spawn special project and inform - RESPAWN
    is_triggered_only = yes
    title = "voidark_vault_invasion.1.title"
    desc = {
        trigger = {
            fromfrom = {
                NOR = {
                    has_country_flag = acot_phanon_tech_stage_one_enabled
                    has_country_flag = acot_phanon_tech_stage_two_enabled
                    has_country_flag = acot_phanon_tech_stage_three_enabled
                    has_country_flag = acot_phanon_tech_stage_four_enabled
                }
            }
        }
        text = "voidark_vault_invasion.1.stage1.desc"
    }
    desc = {
        trigger = {
            OR = {
                has_country_flag = acot_phanon_tech_stage_one_enabled
                has_country_flag = acot_phanon_tech_stage_two_enabled
            }
        }
        text = "voidark_vault_invasion.1.stage2_or_3.desc"
    }
    desc = {
        trigger = {
            has_country_flag = acot_phanon_tech_stage_three_enabled
        }
        text = "voidark_vault_invasion.1.stage4.desc"
    }
    picture = GFX_evt_arkship_combat
    show_sound = event_arkship_in_crisis
    trigger = {
        NOT = { has_global_flag = aot_phanon_content_OFF }
        NOR = {
            has_country_flag = acot_phanon_tech_stage_four_enabled
            has_country_flag = vault_invasion_project_delayed
        }
        is_nomadic = yes
        is_ai = no
        OR = {
            is_same_value = event_target:acot_phanon_original_aggressor
            is_same_value = event_target:acot_phanon_real_spook_country
        }
    }

    option = {
        name = voidark_vault_invasion.1.a
        custom_tooltip = voidark_vault_invasion_enable_project_tt
        hidden_effect = {
            enable_special_project = {
                name = VOIDARK_PHANON_VAULT_INVASION
                owner = event_target:vault_project_receiver
                location = event_target:vault_location
            }
        }
    }

    option = {
        name = voidark_vault_invasion.1.b
        custom_tooltip = voidark_vault_invasion_delay_tt
        hidden_effect = {
            fromfrom = {
                set_timed_country_flag = {
                    flag = vault_invasion_project_delayed
                    days = 179
                }
                country_event = {
                    id = voidark_vault_invasion.11
                    days = 180
                }
            }
        }
    }
}

        }
    }
}
