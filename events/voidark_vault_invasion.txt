    is_triggered_only = yes

    immediate = {

            id = voidark_vault_invasion.991
            months = 1 #delay to allow passage of time
        }
    }
}

#on prior event - set up global events & flags
country_event = {
    id = voidark_vault_invasion.991
    is_triggered_only = yes
    hide_window = yes

    trigger = {
        is_nomadic = yes
        is_ai = no
        NOT = { has_global_flag = acot_phanon_forbidden }
        NOT = { has_global_flag = aot_phanon_content_OFF }
    }
    immediate = {
        owner? = {
            save_global_event_target_as = vault_temp_owner
            set_country_flag = vault_temp_owner_flag
            capital_scope = {
                save_global_event_target_as = worldark_colony
            }
        }
        random_system = {
            limit = {
                has_star_flag = phanon_base_system
            }
            save_global_event_target_as = phanon_system
            every_system_planet = {
                limit = {
                    OR = {
                        is_planet_class = pc_acot_pmc_vault_unshielded
                        is_planet_class = pc_acot_pmc_vault
                    }
                }
                save_global_event_target_as = voidark_phanon_vault
                set_carrier_flag = nomad_recently_settled_flag
            }
        }
    }
}

            name = VOIDARK_PHANON_VAULT_INTEGRATION_PERMISSION
            location = event_target:voidark_phanon_vault
            owner = event_target:vault_temp_owner
        }
    }

    #KEEP VAULT
    option = {
        name = voidark_vault_invasion.993.b
        custom_tooltip = voidark_vault_invasion.993.b.tt
    }
}

#resettle and changing planet
carrier_event = {
    id = voidark_vault_invasion.998
    is_triggered_only = yes
    hide_window = yes
    trigger = { always = yes }
    immediate = {
        event_target:voidark_normal_country = {
            country_event = {
                id = voidark_vault_invasion.999
                days = 30
            }
        }
        every_owned_pop_group = {
            resettle_pop_group = {
                POP_GROUP = THIS
                PLANET = event_target:worldark_colony
                PERCENTAGE = 100
            }
        }
        change_pc = pc_voidark_abandoned_vault
        if = {
            limit = {
                AND = {
                    solar_system = {
                        any_system_megastructure = {
                            has_megastructure_flag = acot_pmc_star_traveller_locked
                        }
                    }
                }
            }
            event_target:voidark_rival_country = {
                set_country_flag = voidark_hacked_star_traveller
            }
        }
    }
}

#Integration of data & storage

country_event = {
    id = voidark_vault_invasion.999
    is_triggered_only = yes
    trigger = {
        is_same_value = event_target:voidark_normal_country
        NOT = { is_same_value = event_target:voidark_rival_country }
        has_integrated_all_data_and_storage_planets = no
    }
    title = voidark_vault_invasion.999.title
    desc = {
        text = voidark_vault_invasion.999.desc
        trigger = {
            AND = {
                is_same_value = event_target:voidark_normal_country
                NOT = { is_same_value = event_target:voidark_rival_country }
            }
        }
    }
    picture = GFX_evt_arkship_in_orbit #todo change?
    option = {
        name = voidark_vault_invasion.999.a #Begin integration & removal of system
        custom_tooltip = voidark_vault_invasion.999.a.tt
        hidden_effect = {
            set_country_flag = voidark_remove_droppoint_resources
            random_system = {
                limit = {
                    has_star_flag = phanon_base_system
                }
                every_system_planet = {
                    limit = {
                        AND = {
                            OR = {
                                is_planet_class = pc_acot_pmc_storage_planet
                                is_planet_class = pc_acot_pmc_data_planet
                            }
                            NOT = { has_carrier_flag = voidark_deposit_removed }
                        }
                    }
                    enable_special_project = {
                        name = VOIDARK_PHANON_RESOURCE_PLANET_INTEGRATION
                        owner = event_target:vault_temp_owner
                        location = this
                    }
                }
            }
        }
    }

    option = {
        name = voidark_vault_invasion.999.b
        custom_tooltip = voidark_vault_invasion.999.b.tt
        hidden_effect = {
            set_country_flag = voidark_keep_droppoint_resources
            end_event_chain = voidark_vault_integration
        }
    }
}
