country_event = {
    id = voidark_vault_invasion.991
    is_triggered_only = yes
    hide_window = yes

    trigger = {
        is_nomadic = yes
        is_ai = no
        NOT = { has_global_flag = acot_phanon_forbidden }
        NOT = { has_global_flag = aot_phanon_content_OFF }
    }
    immediate = {
        owner? = {
            save_global_event_target_as = vault_temp_owner
            set_country_flag = vault_temp_owner_flag
        }
        random_system = {
            limit = {
                has_star_flag = phanon_base_system
            }
            every_system_planet = {
                limit = {
                    OR = {
                        is_planet_class = pc_acot_pmc_vault_unshielded
                        is_planet_class = pc_acot_pmc_vault
                    }
                }
                set_carrier_flag = nomad_recently_settled_flag
            }
        }
    }
}

system_event = {
    id = voidark_vault_invasion.1
    is_triggered_only = yes

    trigger = {
        owner? = {
            acot_is_phanon_country = yes
        }
        from = {
            is_nomadic = yes
            NOT = { has_waystation_pact = root.owner }
        }
    }

    immediate = {
        fromfrom = {
            save_event_target_as = interloper_waystation
        }
        from = {
            save_event_target_as = interloper
        }
        if = {
            limit = {
                event_target:interloper = {
                    NOT = { has_country_flag = voidark_phanon_nomad_tolerance_first_offense }
                }
            }
            event_target:interloper = {
                set_country_flag = voidark_phanon_nomad_tolerance_first_offense
                country_event = {
                    id = voidark_vault_invasion.11
                }
            }
            hidden_effect = {
                destroy_ship = event_target:interloper_waystation
            }
        }
        else_if = {
            limit = {
                event_target:interloper = {
                    has_country_flag = voidark_phanon_nomad_tolerance_first_offense
                }
            }
            event_target:interloper = {
                set_country_flag = voidark_phanon_nomad_tolerance_final
                country_event = {
                    id = voidark_vault_invasion.11
                }
            }
            hidden_effect = {
                destroy_ship = event_target:interloper_waystation
            }
            this.owner = {
                set_closed_borders = {
                    who = event_target:interloper
                    status = yes
                }
            }
        }
    }
}
